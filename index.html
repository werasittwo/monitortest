  2. กดปุ่ม "คัดลอกโค้ด Apps Script" ด้านล่างนี้ ไปวางทับในไฟล์ Code.gs แล้วกด การทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment) เลือกประเภทเป็น เว็บแอป (Web app) ตั้งค่าผู้มีสิทธิ์เข้าถึงเป็น ทุกคน (Anyone)
  3. คัดลอก Web App URL ที่ได้ มาวางในช่องด้านล่างนี้ แล้วกดบันทึก ระบบจะสร้างตารางและซิงค์ข้อมูลทั้งหมดลง Sheet ทันที!
โค้ดสำหรับวางใน Apps Script (อยู่ในไฟล์ google-apps-script/Code.gs ด้วย): 
const SPREADSHEET_ID = '1Lro8WK25uexe97cFHHedqC5iu2B65VyfiQ2AWUXyIUw';
const SHEET_NAME = 'KPI_2570';
const HEADERS = ['id','code','name','dimension','unit','weight','target','actual','polarity','level1','level2','level3','level4','level5','owner','evidenceUrl','evidenceName','note','updatedAt'];

function getOrCreateSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.setName(SHEET_NAME);
    sheet.appendRow(HEADERS);
  }
  return sheet;
}

function readAllRows() {
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0].map(String);
  return data.slice(1).filter(r => r[0] || r[1] || r[2]).map(row => {
    const obj = {};
    headers.forEach((h, i) => obj[h] = row[i]);
    return obj;
  });
}

function writeAllRows(items) {
  const sheet = getOrCreateSheet();
  sheet.clearContents();
  const matrix = [HEADERS, ...items.map(item => HEADERS.map(h => item[h] ?? ''))];
  sheet.getRange(1, 1, matrix.length, HEADERS.length).setValues(matrix);
}

function doGet() {
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: readAllRows() })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  let items = readAllRows();
  if (payload.action === 'replaceAll') items = payload.items || [];
  else if (payload.action === 'upsert') {
    const idx = items.findIndex(x => String(x.id) === String(payload.item.id));
    if (idx >= 0) items[idx] = payload.item; else items.push(payload.item);
  } else if (payload.action === 'delete') {
    items = items.filter(x => String(x.id) !== String(payload.id));
  } else if (payload.action === 'uploadFile') {
    const folders = DriveApp.getFoldersByName('KPI_2570_Evidence_Files');
    const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder('KPI_2570_Evidence_Files');
    const blob = Utilities.newBlob(Utilities.base64Decode(payload.base64Data), payload.mimeType, payload.fileName);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    const idx = items.findIndex(x => String(x.id) === String(payload.kpiId));
    if (idx >= 0) { items[idx].evidenceUrl = file.getUrl(); items[idx].evidenceName = payload.fileName; }
    writeAllRows(items);
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', fileUrl: file.getUrl(), data: items })).setMimeType(ContentService.MimeType.JSON);
  }
  writeAllRows(items);
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', data: items })).setMimeType(ContentService.MimeType.JSON);
}
